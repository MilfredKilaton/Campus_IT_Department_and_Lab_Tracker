using System; using System.Collections.Generic; using CampusAssetTracker.Core.Exceptions; using CampusAssetTracker.Core.Interfaces; using CampusAssetTracker.Core.Models; using CampusAssetTracker.Core.Services; namespace CampusAssetTracker.Console { public class Program { private static readonly AssetService Service = new(); private static readonly string[] MenuOptions = { "List all assets", "Add asset", "Check out item", "Check in item", "Send to repair / Return from repair", "Search assets by name", "Filter by status", "Show checked-out items (LINQ)", "Show maintenance status report", "Total depreciation", "Show assets of a specific type", "Exit" }; private static void Main() { bool running = true; while (running) { int choice = DrawMainMenu(); switch (choice) { case 0: ListAll(); break; case 1: AddAsset(); break; case 2: CheckOut(); break; case 3: CheckIn(); break; case 4: RepairMenu(); break; case 5: Search(); break; case 6: FilterStatus(); break; case 7: ShowCheckedOut(); break; case 8: ShowMaintenanceReport(); break; case 9: ShowDepreciation(); break; case 10: FilterByType(); break; case 11: case -1: running = false; break; } } } private static int DrawMainMenu() { int selected = 0; int width = 0; foreach (var opt in MenuOptions) width = Math.Max(width, opt.Length); width = Math.Max(width, "===== Campus IT Asset Tracker =====".Length); const int padding = 4; int inner = width + padding; string top = "┌" + new string('─', inner) + "┐"; string mid = "├" + new string('─', inner) + "┤"; string bottom = "└" + new string('─', inner) + "┘"; string title = "│ " + "===== Campus IT Asset Tracker =====".PadRight(width + 2) + "│"; while (true) { System.Console.Clear(); System.Console.WriteLine(top); System.Console.WriteLine(title); System.Console.WriteLine(mid); for (int i = 0; i < MenuOptions.Length; i++) { bool isSelected = i == selected; string prefix = isSelected ? "► " : " "; string label = $"{prefix}{i + 1}. {MenuOptions[i]}"; string line = "│ " + label.PadRight(inner - 1) + "│"; if (isSelected) { System.Console.BackgroundColor = ConsoleColor.DarkCyan; System.Console.ForegroundColor = ConsoleColor.White; System.Console.WriteLine(line); System.Console.ResetColor(); } else { System.Console.WriteLine(line); } } System.Console.WriteLine(bottom); System.Console.WriteLine(); System.Console.WriteLine("↑/↓: Navigate Enter: Select Esc/Q: Exit"); var key = System.Console.ReadKey(intercept: true).Key; switch (key) { case ConsoleKey.UpArrow: selected = (selected - 1 + MenuOptions.Length) % MenuOptions.Length; break; case ConsoleKey.DownArrow: selected = (selected + 1) % MenuOptions.Length; break; case ConsoleKey.Enter: return selected; case ConsoleKey.Escape: case ConsoleKey.Q: return -1; } } } private static void ListAll() { System.Console.Clear(); var all = Service.GetAll(); if (all.Count == 0) { System.Console.WriteLine("No assets found."); } else { foreach (var a in all) System.Console.WriteLine(a); } Pause(); } private static void AddAsset() { System.Console.Clear(); System.Console.WriteLine("=== Add Asset ==="); System.Console.WriteLine("1 = Workstation"); System.Console.WriteLine("2 = Networking Hardware"); System.Console.WriteLine("3 = Peripheral"); System.Console.WriteLine("4 = Software License"); System.Console.Write("Type: "); var type = System.Console.ReadLine(); System.Console.Write("Asset Tag: "); var tag = System.Console.ReadLine() ?? string.Empty; System.Console.Write("Name: "); var name = System.Console.ReadLine() ?? string.Empty; System.Console.Write("Purchase Price: "); var priceInput = System.Console.ReadLine(); if (!decimal.TryParse(priceInput, out var price) || price < 0) { System.Console.WriteLine("Invalid price."); Pause(); return; } try { CampusAsset asset = type switch { "1" => new Workstation(tag, name, price, DateTime.UtcNow, "Generic CPU"), "2" => new NetworkingHardware(tag, name, price, DateTime.UtcNow), "3" => new Peripheral(tag, name, price, DateTime.UtcNow), "4" => new SoftwareLicense(tag, name, price, DateTime.UtcNow, 12, DateTime.UtcNow.AddMonths(12)), _ => throw new AssetException("Unknown asset type.") }; Service.Add(asset); System.Console.WriteLine("Asset added successfully."); } catch (AssetException ex) { System.Console.WriteLine($"Error: {ex.Message}"); } Pause(); } private static void CheckOut() { System.Console.Clear(); System.Console.WriteLine("=== Check Out ==="); System.Console.Write("Asset Tag: "); var tag = System.Console.ReadLine() ?? string.Empty; System.Console.Write("Borrower ID: "); var borrower = System.Console.ReadLine() ?? string.Empty; try { Service.CheckOut(tag, borrower); System.Console.WriteLine($"Asset checked out to {borrower}."); } catch (AssetException ex) { System.Console.WriteLine($"Error: {ex.Message}"); } Pause(); } private static void CheckIn() { System.Console.Clear(); System.Console.WriteLine("=== Check In ==="); System.Console.Write("Asset Tag: "); var tag = System.Console.ReadLine() ?? string.Empty; try { Service.CheckIn(tag); System.Console.WriteLine($"Asset checked in."); } catch (AssetException ex) { System.Console.WriteLine($"Error: {ex.Message}"); } Pause(); } private static void RepairMenu() { System.Console.Clear(); System.Console.WriteLine("=== Repair Menu ==="); System.Console.Write("Asset Tag: "); var tag = System.Console.ReadLine() ?? string.Empty; System.Console.Write("(1) Send to repair (2) Return from repair: "); var action = System.Console.ReadLine(); try { if (action == "1") { Service.SendToRepair(tag); System.Console.WriteLine($"Asset sent to repair."); } else if (action == "2") { Service.ReturnFromRepair(tag); System.Console.WriteLine($"Asset returned from repair."); } else { System.Console.WriteLine("Invalid action."); } } catch (AssetException ex) { System.Console.WriteLine($"Error: {ex.Message}"); } Pause(); } private static void Search() { System.Console.Clear(); System.Console.WriteLine("=== Search by Name ==="); System.Console.Write("Keyword: "); var keyword = System.Console.ReadLine() ?? string.Empty; var results = Service.SearchByName(keyword); if (results.Count == 0) { System.Console.WriteLine("No matching assets."); } else { foreach (var a in results) System.Console.WriteLine(a); } Pause(); } private static void FilterStatus() { System.Console.Clear(); System.Console.WriteLine("=== Filter by Status ==="); System.Console.WriteLine("Options: Available, InUse, InRepair, Retired"); System.Console.Write("Status: "); var input = System.Console.ReadLine() ?? string.Empty; if (!Enum.TryParse(input, true, out var status)) { System.Console.WriteLine("Invalid status."); Pause(); return; } var results = Service.FilterByStatus(status); if (results.Count == 0) { System.Console.WriteLine("No assets with that status."); } else { foreach (var a in results) System.Console.WriteLine(a); } Pause(); } private static void FilterByType() { System.Console.Clear(); System.Console.WriteLine("=== Filter by Type ==="); System.Console.WriteLine("1 = Workstation"); System.Console.WriteLine("2 = Networking Hardware"); if (results.Count == 0) { System.Console.WriteLine("No assets found for that type."); } else { foreach (var a in results) System.Console.WriteLine(a); } Pause(); } private static void ShowCheckedOut() { System.Console.Clear(); } else { foreach (var item in checkedOut) { if (item is CampusAsset asset) System.Console.WriteLine($"[{asset.AssetTag}] {asset.Name} - Borrower: {item.BorrowerId}"); else System.Console.WriteLine($"Borrower: {item.BorrowerId}"); } } Pause(); } private static void ShowMaintenanceReport() { System.Console.Clear(); System.Console.WriteLine("=== Maintenance Status Report (Polymorphism) ==="); foreach (var a in Service.GetAll()) System.Console.WriteLine($"{a.AssetTag,-10} {a.Name,-30} -> {a.GetMaintenanceStatus()}"); Pause(); } private static void ShowDepreciation() { System.Console.Clear(); System.Console.WriteLine("=== Total Depreciation (Polymorphism) ==="); System.Console.WriteLine($"Total current value of all assets: {Service.GetTotalDepreciation():C}"); Pause(); } private static void Pause() { System.Console.WriteLine(); System.Console.WriteLine("Press any key to return to the menu..."); System.Console.ReadKey(intercept: true); } } }
